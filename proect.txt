#include <iostream>
#include <vector>
#include <deque>
#include <cmath>
#include <numeric>

// ===============================================================
// КЛАСС ДЛЯ ВАЛИДАЦИИ ПОВЕДЕНЧЕСКОЙ ЦЕЛОСТНОСТИ БТС
// Обнаружение медленных кибератак через анализ отклонения от плана
// ===============================================================
class BehavioralValidator {
private:
    // ------------------- ПАРАМЕТРЫ ПОРОГОВ -------------------
    double DEVIATION_THRESHOLD = 0.5;   // Порог отклонения в метрах (0.5 м = 50 см)
                                        // Если среднее отклонение превышает этот порог
                                        // и автомобиль формально соблюдает правила,
                                        // то это признак медленной атаки
    
    // ------------------- ХРАНЕНИЕ ИСТОРИИ -------------------
    std::deque<double> deviation_history;   // Очередь для хранения истории отклонений
                                            // Используем deque (двусторонняя очередь), 
                                            // чтобы быстро добавлять новые значения
                                            // и удалять старые, когда их становится слишком много
    
    bool attack_triggered = false;          // Флаг, который становится true ПРИ ПЕРВОМ обнаружении атаки
                                            // После этого никогда не сбрасывается
    
    size_t max_history_size = 50;           // Максимальный размер истории (50 измерений)
                                            // При частоте 20 Гц (0.05 сек) это покрывает 2.5 секунды
    
    // ------------------- СТАТИСТИЧЕСКИЕ ПЕРЕМЕННЫЕ -------------------
    double sum_deviation = 0.0;             // Сумма всех отклонений в истории
                                            // Используется для быстрого вычисления среднего
                                            // Вместо перебора всей очереди при каждом вызове
    
public:
    // ===============================================================
    // ГЛАВНАЯ ФУНКЦИЯ ВАЛИДАЦИИ
    // Входные параметры:
    //   current_time - текущее время (секунды)
    //   actual_y     - фактическое поперечное положение автомобиля (метры)
    //   planned_y    - плановое поперечное положение (что предписывает планировщик)
    // ===============================================================
    void validate(double current_time, double actual_y, double planned_y) {
        
        // ---------------- ШАГ 1: ВЫЧИСЛЕНИЕ МГНОВЕННОГО ОТКЛОНЕНИЯ ----------------
        // Мгновенное отклонение = модуль разницы между фактом и планом
        // Чем больше это значение, тем сильнее автомобиль отклоняется от заданной траектории
        double instant_deviation = std::abs(actual_y - planned_y);
        
        // ---------------- ШАГ 2: ОБНОВЛЕНИЕ СУММЫ ОТКЛОНЕНИЙ ----------------
        // Добавляем новое отклонение в сумму
        sum_deviation += instant_deviation;
        
        // Добавляем новое отклонение в историю
        deviation_history.push_back(instant_deviation);
        
        // ---------------- ШАГ 3: УДАЛЕНИЕ СТАРЫХ ДАННЫХ ----------------
        // Если история превысила максимальный размер,
        // удаляем самое старое значение и вычитаем его из суммы
        if (deviation_history.size() > max_history_size) {
            double oldest_deviation = deviation_history.front();  // Берём самое старое значение
            deviation_history.pop_front();                         // Удаляем его из очереди
            sum_deviation -= oldest_deviation;                     // Вычитаем из общей суммы
        }
        
        // ---------------- ШАГ 4: РАСЧЁТ СРЕДНЕГО ОТКЛОНЕНИЯ ----------------
        // Среднее арифметическое = сумма всех отклонений / количество измерений
        // Это даёт нам сглаженное значение, устойчивое к случайным шумам
        double mean_deviation = 0.0;
        
        if (deviation_history.size() > 0) {
            mean_deviation = sum_deviation / deviation_history.size();
        }
        
        // ------------------- ШАГ 5: ОТЛАДОЧНЫЙ ВЫВОД -------------------
        // Выводим статистику каждые 50 итераций (примерно каждые 2.5 секунды при dt=0.05)
        static int call_counter = 0;
        if (++call_counter % 50 == 0) {
            // Подробный вывод для отладки
            std::cout << "[DEBUG] Мгновенное отклонение: " << instant_deviation << " м";
            std::cout << " | Среднее отклонение: " << mean_deviation << " м";
            std::cout << " | Факт Y: " << actual_y << " м";
            std::cout << " | План Y: " << planned_y << " м";
            std::cout << " | Размер истории: " << deviation_history.size() << std::endl;
        }
        
        // ---------------- ШАГ 6: ОБНАРУЖЕНИЕ АТАКИ ----------------
        // Если среднее отклонение превышает порог И атака ещё не была обнаружена
        if (mean_deviation > DEVIATION_THRESHOLD && !attack_triggered) {
            
            // Устанавливаем флаг атаки
            attack_triggered = true;
            
            // Выводим сообщение об обнаружении
            std::cout << "\n==================================================" << std::endl;
            std::cout << "[ALERT] МЕДЛЕННАЯ АТАКА ОБНАРУЖЕНА!" << std::endl;
            std::cout << "==================================================" << std::endl;
            std::cout << "   Среднее отклонение: " << mean_deviation << " м" << std::endl;
            std::cout << "   Порог обнаружения: " << DEVIATION_THRESHOLD << " м" << std::endl;
            std::cout << "   Фактическое положение Y: " << actual_y << " м" << std::endl;
            std::cout << "   Плановое положение Y: " << planned_y << " м" << std::endl;
            std::cout << "   Расхождение с планом: " << std::abs(actual_y - planned_y) << " м" << std::endl;
            std::cout << "   Размер окна истории: " << deviation_history.size() << " измерений" << std::endl;
            
            // Вызываем защитные механизмы
            activateFallback();
        }
    }
    
    // ===============================================================
    // ФУНКЦИЯ АКТИВАЦИИ ЗАЩИТНЫХ МЕХАНИЗМОВ (FALLBACK)
    // Вызывается при обнаружении атаки
    // ===============================================================
    void activateFallback() {
        std::cout << "\n[FALLBACK] Активация усиленной защиты!" << std::endl;
        std::cout << "   ✓ Снижение скорости до безопасного значения" << std::endl;
        std::cout << "   ✓ Переход на резервный планировщик траектории" << std::endl;
        std::cout << "   ✓ Уведомление удалённого оператора" << std::endl;
        std::cout << "   ✓ Логирование инцидента для дальнейшего анализа" << std::endl;
        std::cout << "==================================================\n" << std::endl;
    }
    
    // ===============================================================
    // ПРОВЕРКА, БЫЛА ЛИ ОБНАРУЖЕНА АТАКА
    // ===============================================================
    bool isAttackDetected() const { 
        return attack_triggered; 
    }
    
    // ===============================================================
    // ПОЛУЧИТЬ ТЕКУЩЕЕ СРЕДНЕЕ ОТКЛОНЕНИЕ (для отладки)
    // ===============================================================
    double getCurrentMeanDeviation() const {
        if (deviation_history.empty()) return 0.0;
        return sum_deviation / deviation_history.size();
    }
    
    // ===============================================================
    // ПОЛУЧИТЬ РАЗМЕР ИСТОРИИ (для отладки)
    // ===============================================================
    size_t getHistorySize() const {
        return deviation_history.size();
    }
};

// ===============================================================
// ГЛАВНАЯ ФУНКЦИЯ - СИМУЛЯЦИЯ ДВИЖЕНИЯ БТС
// ===============================================================
int main() {
    std::cout << "\n========================================" << std::endl;
    std::cout << "Behavioral Integrity Validator для БТС" << std::endl;
    std::cout << "Обнаружение медленных кибератак" << std::endl;
    std::cout << "Расчёт среднего отклонения от плана" << std::endl;
    std::cout << "========================================\n" << std::endl;
    
    // ========== ТЕСТ 1: Нормальное движение (без атак) ==========
    std::cout << "[ТЕСТ 1] Нормальное движение (атак нет)" << std::endl;
    std::cout << "=========================================" << std::endl;
    
    BehavioralValidator validator1;
    double time = 0.0;
    double actual_y = 0.0;    // Фактическое положение (Y координата)
    double planned_y = 0.0;   // Плановое положение (всегда 0)
    const double dt = 0.05;    // Шаг времени 50 мс (20 Гц)
    
    // Симулируем 25 секунд движения
    for (int step = 0; step < 500; step++) {
        time += dt;
        
        // Нормальное движение: автомобиль точно следует плану
        actual_y = 0.0;      // Никаких отклонений
        planned_y = 0.0;
        
        // Валидация текущего состояния
        validator1.validate(time, actual_y, planned_y);
        
        // Если атака обнаружена (не должно случиться) - выходим
        if (validator1.isAttackDetected()) break;
    }
    
    // Вывод результата первого теста
    std::cout << "\n=== ИТОГИ ТЕСТА 1 ===" << std::endl;
    std::cout << "Атака обнаружена: " << (validator1.isAttackDetected() ? "ДА" : "НЕТ") << std::endl;
    std::cout << "Среднее отклонение: " << validator1.getCurrentMeanDeviation() << " м" << std::endl;
    
    if (!validator1.isAttackDetected()) {
        std::cout << "ИТОГ: ✓ Атак не обнаружено (система работает корректно)" << std::endl;
    } else {
        std::cout << "ИТОГ: ✗ ЛОЖНОЕ СРАБАТЫВАНИЕ!" << std::endl;
    }
    
    // ========== ТЕСТ 2: Медленная атака (постепенное смещение) ==========
    std::cout << "\n\n[ТЕСТ 2] Медленная атака: постепенное смещение с полосы" << std::endl;
    std::cout << "=========================================================" << std::endl;
    
    BehavioralValidator validator2;
    time = 0.0;
    actual_y = 0.0;
    planned_y = 0.0;
    
    double lateral_velocity = 0.0;      // Поперечная скорость (м/с)
    double attack_intensity = 0.03;     // Интенсивность атаки (м/с²) - каждую секунду скорость растёт на 0.03 м/с
    
    // Симулируем до 40 секунд или до обнаружения атаки
    for (int step = 0; step < 800; step++) {
        time += dt;
        
        // МЕДЛЕННАЯ АТАКА: постепенное ускорение в поперечном направлении
        // Злоумышленник незаметно изменяет управляющие сигналы,
        // заставляя автомобиль медленно съезжать с полосы
        lateral_velocity += attack_intensity * dt;   // Увеличиваем поперечную скорость
        actual_y += lateral_velocity * dt;            // Обновляем поперечное положение
        
        // План НЕ ЗНАЕТ об атаке - он всё ещё предписывает ехать прямо
        planned_y = 0.0;
        
        // Если отклонение стало заметным - выводим предупреждение
        if (std::abs(actual_y) > 0.5) {
            std::cout << "[SIM] Текущее отклонение: " << actual_y << " м от полосы" << std::endl;
        }
        
        // ВАЛИДАЦИЯ: проверяем, не атака ли это
        validator2.validate(time, actual_y, planned_y);
        
        // Если атака обнаружена - завершаем симуляцию
        if (validator2.isAttackDetected()) break;
    }
    
    // Вывод результата второго теста
    std::cout << "\n=== ИТОГИ ТЕСТА 2 ===" << std::endl;
    std::cout << "Атака обнаружена: " << (validator2.isAttackDetected() ? "ДА" : "НЕТ") << std::endl;
    std::cout << "Финальное отклонение: " << actual_y << " м" << std::endl;
    std::cout << "Среднее отклонение (окно " << validator2.getHistorySize() << " измерений): " 
              << validator2.getCurrentMeanDeviation() << " м" << std::endl;
    
    if (validator2.isAttackDetected()) {
        std::cout << "ИТОГ: ✓ Медленная атака УСПЕШНО ОБНАРУЖЕНА!" << std::endl;
    } else {
        std::cout << "ИТОГ: ✗ Атака не обнаружена (ошибка)" << std::endl;
    }
    
    return 0;
}
